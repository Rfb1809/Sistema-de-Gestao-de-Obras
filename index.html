function doGet(e) { var params = e.parameter; var userProcurado = params.usuario; var senhaProcurada = params.senha; var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("BD_Usuarios_Sistema"); if (!sheet) { return ContentService.createTextOutput(JSON.stringify({ "status": "erro", "mensagem": "Nenhum usuario cadastrado ainda." })).setMimeType(ContentService.MimeType.JSON); } var dados = sheet.getDataRange().getValues(); for (var i = 1; i < dados.length; i++) { if (dados[i][0] == userProcurado && dados[i][1] == senhaProcurada) { return ContentService.createTextOutput(JSON.stringify({ "status": "sucesso", "perm": dados[i][2] })).setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput(JSON.stringify({ "status": "erro", "mensagem": "Usuario ou senha incorretos." })).setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); // AÇÃO ESPECÍFICA DE CRIAÇÃO DE USUÁRIOS NA NUVEM if (data.acao === "criarUsuario") { var sheetUsers = ss.getSheetByName("BD_Usuarios_Sistema"); if (!sheetUsers) { sheetUsers = ss.insertSheet("BD_Usuarios_Sistema"); sheetUsers.appendRow(["Usuario", "Senha", "Permissao"]); } sheetUsers.appendRow([data.username, data.password, data.perm]); return ContentService.createTextOutput(JSON.stringify({"status": "sucesso"})).setMimeType(ContentService.MimeType.JSON); } // PROCESSO NORMAL DE SALVAMENTO DE OBRAS E MANUTENÇÃO var sheetObras = ss.getSheetByName("BD_Obras_Macro") || ss.insertSheet("BD_Obras_Macro"); var sheetClientes = ss.getSheetByName("BD_Manutencao_Plantas") || ss.insertSheet("BD_Manutencao_Plantas"); sheetObras.clear(); sheetClientes.clear(); sheetObras.appendRow(["ID", "Dados_JSON"]); if(data.obras) { data.obras.forEach(function(o) { sheetObras.appendRow([o.id, JSON.stringify(o)]); }); } sheetClientes.appendRow(["ID", "Dados_JSON"]); if(data.clientes) { data.clientes.forEach(function(c) { sheetClientes.appendRow([c.id, JSON.stringify(c)]); }); } return ContentService.createTextOutput(JSON.stringify({"status": "sucesso"})).setMimeType(ContentService.MimeType.JSON); }
